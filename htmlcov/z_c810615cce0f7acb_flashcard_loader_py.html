1""" 

2Load and validate flashcard decks from JSON files. 

3 

4This module is the data layer for the quiz app. It reads a deck file, 

5accepts either of the two supported JSON shapes (a bare array of cards 

6or an object with a "cards" key), and rejects anything malformed with a 

7clear, human-readable message instead of a raw Python traceback. 

8""" 

9 

10import json 

11from dataclasses import dataclass 

12from pathlib import Path 

13from typing import Any 

14 

15 

16class FlashcardError(Exception): 

17 """Raised when a deck file cannot be read or is not valid. 

18 

19 The message is meant to be shown directly to the user, so it must 

20 never contain a stack trace or internal jargon. 

21 """ 

22 

23 

24@dataclass(frozen=True) 

25class Flashcard: 

26 """A single question/answer pair. 

27 

28 ``frozen=True`` makes instances immutable: once a card is loaded, 

29 nothing elsewhere in the program can change it by accident. 

30 """ 

31 

32 front: str 

33 back: str 

34 

35 

36def load_flashcards(filepath: str) -> list[Flashcard]: 

37 """Read a deck file and return a list of ``Flashcard`` objects. 

38 

39 Args: 

40 filepath: Path to a ``.json`` deck file. 

41 

42 Returns: 

43 A non-empty list of validated flashcards. 

44 

45 Raises: 

46 FlashcardError: If the file is missing, is not valid JSON, does 

47 not match a supported shape, or contains an invalid card. 

48 """ 

49 path = Path(filepath) 

50 

51 try: 

52 raw_text = path.read_text(encoding="utf-8") 

53 except FileNotFoundError as exc: 

54 raise FlashcardError(f"Could not find a deck file at: {filepath}") from exc 

55 except OSError as exc: 

56 raise FlashcardError(f"Could not read {filepath}: {exc}") from exc 

57 

58 try: 

59 data = json.loads(raw_text) 

60 except json.JSONDecodeError as exc: 

61 raise FlashcardError( 

62 f"{filepath} is not valid JSON (line {exc.lineno}, column {exc.colno})." 

63 ) from exc 

64 

65 return _parse_cards(data, filepath) 

66 

67 

68def _parse_cards(data: Any, source: str) -> list[Flashcard]: 

69 """Turn already-parsed JSON data into a list of flashcards. 

70 

71 Kept separate from file reading so the validation rules can be 

72 tested without touching the filesystem. 

73 """ 

74 if isinstance(data, dict): 

75 if "cards" not in data: 

76 raise FlashcardError(f"{source} is a JSON object but has no 'cards' key.") 

77 raw_cards = data["cards"] 

78 elif isinstance(data, list): 

79 raw_cards = data 

80 else: 

81 raise FlashcardError( 

82 f"{source} must be a JSON array or an object with a " 

83 f"'cards' key, not {type(data).__name__}." 

84 ) 

85 

86 if not isinstance(raw_cards, list): 

87 raise FlashcardError(f"The 'cards' value in {source} must be a list.") 

88 

89 if not raw_cards: 

90 raise FlashcardError(f"{source} does not contain any flashcards.") 

91 

92 cards: list[Flashcard] = [] 

93 for position, item in enumerate(raw_cards, start=1): 

94 cards.append(_build_card(item, position, source)) 

95 return cards 

96 

97 

98def _build_card(item: Any, position: int, source: str) -> Flashcard: 

99 """Validate one raw card and return a ``Flashcard``.""" 

100 if not isinstance(item, dict): 

101 raise FlashcardError( 

102 f"Card {position} in {source} must be an object, " 

103 f"not {type(item).__name__}." 

104 ) 

105 

106 for field in ("front", "back"): 

107 if field not in item: 

108 raise FlashcardError( 

109 f"Card {position} in {source} is missing the " 

110 f"required field '{field}'." 

111 ) 

112 if not isinstance(item[field], str) or not item[field].strip(): 

113 raise FlashcardError( 

114 f"Card {position} in {source} has an empty or " 

115 f"non-text '{field}' value." 

116 ) 

117 

118 return Flashcard(front=item["front"].strip(), back=item["back"].strip())
